1. AUTHENTICATION
APIs xác thực và quản lý JWT tokens
==================================================

--------------------------------------------------
POST /auth/login
--------------------------------------------------
Tác dụng:
Đăng nhập hệ thống bằng username và password.
Trả về access token và refresh token để sử dụng cho các API bảo mật.

Request:
{
  "username": "string",
  "password": "string"
}

Response 200:
{
  "success": true,
  "data": {
    "accessToken": "string",
    "refreshToken": "string"
  },
  "message": "Login success",
  "error": null
}


--------------------------------------------------
GET /auth/me
--------------------------------------------------
Tác dụng:
Lấy thông tin người dùng hiện tại đang đăng nhập.
Dùng để xác định role và permission nhằm phân quyền giao diện.

Request:
Header:
Authorization: Bearer {access_token}

Response 200:
{
  "success": true,
  "data": {
    "id": 0,
    "username": "string",
    "fullName": "string",
    "status": 0,
    "roles": [
      {
        "id": 0,
        "name": "string",
        "code": "string",
        "permissions": [
          {
            "id": 1,
            "name": "READ_USER",
            "resource": "USER",
            "action": "READ",
            "description": "Quyền đọc thông tin người dùng",
            "createdAt": "2023-12-01T10:00:00Z",
            "createdBy": "admin"
          }
        ]
      }
    ],
    "createdAt": "2025-12-26T17:10:32.067Z",
    "createdBy": "string",
    "updatedAt": "2025-12-26T17:10:32.067Z",
    "updatedBy": "string"
  },
  "message": "string",
  "error": "string"
}


--------------------------------------------------
POST /auth/refresh
--------------------------------------------------
Tác dụng:
Làm mới access token khi token cũ hết hạn.

Request:
{
  "refreshToken": "string"
}

Response 200:
{
  "success": true,
  "data": {
    "accessToken": "string"
  },
  "message": "Refresh token success",
  "error": null
}


--------------------------------------------------
POST /auth/logout
--------------------------------------------------
Tác dụng:
Đăng xuất hệ thống, vô hiệu hóa token.

Response 200:
{
  "success": true,
  "message": "Logout success",
  "error": null
}

  private async connectWebSocket(): Promise<void> {
    try {
      const token = await this.keycloakService.getToken();
      const socketUrl = `${environment.api_end_point}/ws`;

      console.log('[WebSocket] Đang kết nối đến:', socketUrl);

      this.stompClient = new Client({
        webSocketFactory: () => new SockJS(socketUrl),
        connectHeaders: {
          Authorization: `Bearer ${token}`,
          clientType: 'public',
        },
        reconnectDelay: 0, // Tắt auto-reconnect của STOMP, dùng custom reconnect logic
        heartbeatIncoming: 10000, // 10 giây
        heartbeatOutgoing: 10000, // 10 giây
        onConnect: (frame) => {
          console.log('[WebSocket] ✅ Kết nối thành công:', frame);
          this.isConnectWebsocket = true;
          this.isReconnecting = false;
          if (this.reconnectTimeout) {
            clearTimeout(this.reconnectTimeout);
            this.reconnectTimeout = null;
          }
          if (this.isConnectWebsocket && this.machineId) {
            this.subscribeToResponseTopic();
            this.getWarningMahineStopped();
          }
          this.cdr.markForCheck();
        },
        onStompError: (error) => {
          console.error('[WebSocket] ❌ STOMP error:', error);
          this.cdr.markForCheck();
        },
        onWebSocketClose: (event) => {
          console.warn(
            '[WebSocket] ⚠️ WebSocket đã đóng:',
            event.code,
            event.reason
          );
          // Chỉ reconnect nếu không phải là close code bình thường (1000)
          if (event.code !== 1000) {
            // Nếu đang reconnect, reset flag để có thể reconnect lại
            if (this.isReconnecting) {
              this.isReconnecting = false;
            }
            this.handleReconnect();
          }
          this.cdr.markForCheck();
        },
        onWebSocketError: (error) => {
          console.error('[WebSocket] ❌ WebSocket error:', error);
          // Nếu đang reconnect, reset flag để có thể reconnect lại
          if (this.isReconnecting) {
            this.isReconnecting = false;
          }
          this.handleReconnect();
          this.cdr.markForCheck();
        },
        onDisconnect: (frame) => {
          console.warn('[WebSocket] ⚠️ Disconnected:', frame);
          this.cdr.markForCheck();
        },
      });
      this.stompClient.activate();
    } catch (error) {
      console.error('[WebSocket] ❌ Lỗi khi khởi tạo kết nối:', error);
      this.cdr.markForCheck();
    }
  }

  public requestDataFromServer(dataIds: string[]): Promise<any> {
    return new Promise((resolve, reject) => {
      const correlationId = uuidv4();
      const payload: RequestPayload = {
        correlation_id: correlationId,
        user_request: this.username,
        machine_codes: dataIds,
      };

      //
      this.pendingRequests.set(correlationId, resolve);
      this.stompClient.publish({
        destination: '/app/request-to-read',
        body: JSON.stringify(payload),
      });
    });
  }

  private handleReconnect(): void {
    if (this.isReconnecting) {
      return;
    }
    this.isReconnecting = true;
    console.log(
      '[WebSocket] 🔄 Bắt đầu reconnect sau',
      this.RECONNECT_DELAY,
      'ms...'
    );
    if (this.reconnectTimeout) {
      clearTimeout(this.reconnectTimeout);
    }
    this.reconnectTimeout = setTimeout(async () => {
      try {
        console.log('[WebSocket] 🔄 Đang thử reconnect...');

        // Cleanup client cũ hoàn toàn
        if (this.stompClient) {
          try {
            // Unsubscribe chỉ khi WebSocket còn connected
            if (this.stompSubscription && this.stompClient.connected) {
              try {
                this.stompSubscription.unsubscribe();
              } catch (unsubError) {
                console.warn('[WebSocket] ⚠️ Lỗi khi unsubscribe:', unsubError);
              }
            }
            if (this.dataPerMinuteSubscription && this.stompClient.connected) {
              try {
                this.dataPerMinuteSubscription.unsubscribe();
              } catch (unsubError) {
                console.warn('[WebSocket] ⚠️ Lỗi khi unsubscribe:', unsubError);
              }
            }
            // Clear subscription references
            this.stompSubscription = null as any;
            this.dataPerMinuteSubscription = null as any;
            // Deactivate chỉ khi client đang active
            if (this.stompClient.active) {
              try {
                await this.stompClient.deactivate();
                // Đợi một chút để đảm bảo cleanup hoàn tất
                await new Promise((resolve) => setTimeout(resolve, 100));
              } catch (deactivateError) {
                console.warn(
                  '[WebSocket] ⚠️ Lỗi khi deactivate:',
                  deactivateError
                );
              }
            }
          } catch (cleanupError) {
            console.warn(
              '[WebSocket] ⚠️ Lỗi khi cleanup client cũ:',
              cleanupError
            );
          }
          this.stompClient = null as any;
        }

        // Tạo lại connection hoàn toàn
        await this.connectWebSocket();
      } catch (error) {
        console.error('[WebSocket] ❌ Lỗi khi reconnect:', error);
        this.isReconnecting = false;
        this.handleReconnect();
      }
    }, this.RECONNECT_DELAY);
  }

  private getFileExtension(fileName: string): string {
    if (!fileName) return '.docx';
    const lowerFileName = fileName.toLowerCase();
    if (lowerFileName.endsWith('.pdf')) return '.pdf';
    if (lowerFileName.endsWith('.docx')) return '.docx';
    if (lowerFileName.endsWith('.doc')) return '.doc';
    if (lowerFileName.endsWith('.xlsx')) return '.xlsx';
    return '.docx'; // default

  // Helper function để lấy MIME type từ extension

        return 'application/vnd.openxmlformats-officedocument.wordprocessingml.document';
      case '.doc':

        return 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet';

        return 'application/vnd.openxmlformats-officedocument.wordprocessingml.document';

  }

   * Export template với POST request
   * @param sailorIds - mảng ID
   * @param templateId
   * @param language - 'vietnamese' hoặc 'english'
   * @param fontCase - 'lowercase' hoặc 'uppercase'
   * @returns Promise với response chứa base64 docx hoặc zip
   */
  exportSailorBasedOnTemplate(
    sailorIds: number[],
    templateCode: any,
    language: string = 'vietnamese',
    fontCase: string = 'lowercase',
  ) {
    const params = new URLSearchParams();
    params.append('language', language);
    params.append('case', fontCase);

    return this.baseService.postFileZip(
      `api/sailors/export/${templateCode}?${params.toString()}`,
      sailorIds,
    );
  }


   // Method dành riêng cho download file zip (blob response)
  async postFileZip(url: string, data: any): Promise<any> {
    try {
      let response = await firstValueFrom(
        this.http.post<Blob>(`${this.path}/${url}`, data, {
          headers: new HttpHeaders({
            'Content-Type': 'application/json',
            Accept: 'application/zip, application/octet-stream, */*',
            'Access-Control-Allow-Origin': '*',
            'Accept-language': this.accpetLanguage,
          }),
          responseType: 'blob' as 'json',
          observe: 'response',
        }),
      );

      if (response?.status == 200 && response?.body) {
        this.loader.stop();
        return response.body;
      } else {
        throw new Error(`Error downloading file`);
      }
    } catch (error: any) {
      this.loader.stop();
      if (error.error) {
        if (error.error instanceof Blob) {
          const reader = new FileReader();
          reader.onload = () => {
            try {
              const errorText = JSON.parse(reader.result as string);
              if (errorText.result) {
                this.toast.error(errorText.result.message, 'Thông báo');
              }
            } catch (e) {
              this.toast.error('Có lỗi xảy ra khi tải file', 'Thông báo');
            }
          };
          reader.readAsText(error.error);
        } else if (error.error.result) {
          this.toast.error(error.error.result.message, 'Thông báo');
        } else {
          this.toast.error(
            ` Kết nối không ổn định, vui lòng thử lại`,
            'Thông báo',
          );
        }
      } else {
        this.toast.error(
          ` Kết nối không ổn định, vui lòng thử lại`,
          'Thông báo',
        );
      }
      throw error;
    }
  }
